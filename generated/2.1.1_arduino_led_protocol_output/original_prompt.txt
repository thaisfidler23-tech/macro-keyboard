## Arduino Firmware Implementation

### LED Protocol Handler
```cpp
// Core LED protocol processing:
void processLEDCommand(String command) {
  if (command.startsWith("LED:SET:")) {
    // Parse: LED:SET:<button_id>:<r>:<g>:<b>
    int button = parseButtonId(command);
    uint8_t r = parseColorValue(command, 2);
    uint8_t g = parseColorValue(command, 3);
    uint8_t b = parseColorValue(command, 4);
    
    setButtonColor(button, r, g, b);
    updateLEDStrip();
    
  } else if (command.startsWith("LED:BATCH:")) {
    // Efficient batch color updates
    processBatchColorUpdate(command);
    
  } else if (command.startsWith("LED:EFFECT:")) {
    // Trigger lighting effects
    processEffectCommand(command);
  }
}

void setButtonColor(int buttonId, uint8_t r, uint8_t g, uint8_t b) {
  if (buttonId >= 0 && buttonId < MAX_BUTTONS) {
    ledBuffer[buttonId * 3] = r;
    ledBuffer[buttonId * 3 + 1] = g;
    ledBuffer[buttonId * 3 + 2] = b;
  }
}
```

### Lighting Effects Implementation
```cpp
// Built-in lighting effects:
void updateBreathingEffect(uint8_t effectId) {
  static uint32_t lastUpdate = 0;
  static float phase = 0;
  
  uint32_t now = millis();
  if (now - lastUpdate >= EFFECT_UPDATE_INTERVAL) {
    float brightness = (sin(phase) + 1.0) / 2.0;
    
    for (int i = 0; i < buttonCount; i++) {
      uint8_t r = (baseColors[i * 3] * brightness);
      uint8_t g = (baseColors[i * 3 + 1] * brightness);  
      uint8_t b = (baseColors[i * 3 + 2] * brightness);
      setButtonColor(i, r, g, b);
    }
    
    phase += BREATHING_SPEED;
    if (phase >= 2 * PI) phase = 0;
    
    lastUpdate = now;
    updateLEDStrip();
  }
}

void updateRainbowEffect(uint8_t effectId) {
  static uint32_t lastUpdate = 0;
  static uint8_t hue = 0;
  
  uint32_t now = millis();
  if (now - lastUpdate >= RAINBOW_UPDATE_INTERVAL) {
    for (int i = 0; i < buttonCount; i++) {
      uint8_t buttonHue = hue + (i * 255 / buttonCount);
      setButtonColorHSV(i, buttonHue, 255, 255);
    }
    
    hue += RAINBOW_SPEED;
    lastUpdate = now;
    updateLEDStrip();
  }
}
```

## Hardware Integration Considerations

### Power Management
```cpp
// LED power management and safety:
void checkPowerLimits() {
  uint16_t totalPower = calculateLEDPowerUsage();
  
  if (totalPower > MAX_USB_POWER) {
    // Reduce brightness automatically
    globalBrightness = (MAX_USB_POWER * globalBrightness) / totalPower;
    updateLEDStrip();
  }
}

uint16_t calculateLEDPowerUsage() {
  uint16_t totalPower = 0;
  for (int i = 0; i < buttonCount * 3; i++) {
    totalPower += ledBuffer[i];
  }
  return totalPower * LED_CURRENT_PER_LEVEL / 1000; // mA
}
```

### Timing Optimization
```cpp
// Non-blocking LED updates:
void handleLEDUpdates() {
  static uint32_t lastLEDUpdate = 0;
  
  if (micros() - lastLEDUpdate >= LED_UPDATE_INTERVAL) {
    if (effectsActive) {
      updateActiveEffects();
    }
    
    if (ledBufferChanged) {
      updateLEDStrip();
      ledBufferChanged = false;
    }
    
    lastLEDUpdate = micros();
  }
}
```
