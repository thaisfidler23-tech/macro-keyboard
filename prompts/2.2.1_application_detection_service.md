## Application Detection Implementation

### Cross-Platform Detector
```javascript
// Unified application detection interface:
class ApplicationDetector {
  constructor() {
    this.platform = process.platform;
    this.detector = this.createPlatformDetector();
    this.currentApplication = null;
    this.eventListeners = [];
  }
  
  createPlatformDetector() {
    switch (this.platform) {
      case 'win32':
        return new WindowsApplicationDetector();
      case 'darwin':
        return new MacOSApplicationDetector();
      case 'linux':
        return new LinuxApplicationDetector();
      default:
        throw new Error(`Unsupported platform: ${this.platform}`);
    }
  }
  
  async startMonitoring() {
    await this.detector.initialize();
    this.detector.on('applicationChanged', this.handleApplicationChange.bind(this));
    return this.detector.startMonitoring();
  }
  
  handleApplicationChange(applicationInfo) {
    if (this.isDifferentApplication(applicationInfo)) {
      this.currentApplication = applicationInfo;
      this.emitContextChange(applicationInfo);
    }
  }
}
```

### Windows Implementation
```javascript
// Windows-specific application detection:
class WindowsApplicationDetector extends EventEmitter {
  constructor() {
    super();
    this.user32 = ffi.Library('user32', {
      'GetForegroundWindow': ['pointer', []],
      'GetWindowTextA': ['int', ['pointer', 'char*', 'int']],
      'GetWindowThreadProcessId': ['uint32', ['pointer', 'uint32*']]
    });
  }
  
  async getCurrentApplication() {
    const hwnd = this.user32.GetForegroundWindow();
    if (!hwnd.isNull()) {
      const processId = await this.getProcessId(hwnd);
      const windowTitle = await this.getWindowTitle(hwnd);
      const processInfo = await this.getProcessInfo(processId);
      
      return {
        processName: processInfo.name,
        processId: processId,
        windowTitle: windowTitle,
        executablePath: processInfo.path,
        timestamp: Date.now()
      };
    }
    return null;
  }
}
```

### macOS Implementation
```javascript
// macOS-specific application detection:
class MacOSApplicationDetector extends EventEmitter {
  async getCurrentApplication() {
    const frontApp = await this.executeAppleScript(`
      tell application "System Events"
        set frontApp to name of first application process whose frontmost is true
        set frontAppPID to unix id of first application process whose frontmost is true
        return {frontApp, frontAppPID}
      end tell
    `);
    
    if (frontApp) {
      const windowTitle = await this.getActiveWindowTitle();
      
      return {
        processName: frontApp.name,
        processId: frontApp.pid,
        windowTitle: windowTitle,
        bundleId: await this.getBundleId(frontApp.name),
        timestamp: Date.now()
      };
    }
    return null;
  }
  
  async getActiveWindowTitle() {
    return await this.executeAppleScript(`
      tell application "System Events"
        tell process (name of first application process whose frontmost is true)
          try
            set windowTitle to title of front window
            return windowTitle
          on error
            return ""
          end try
        end tell
      end tell
    `);
  }
}
```

### Context Rule Engine
```javascript
// Context rule evaluation system:
class ContextRuleEngine {
  constructor(rules) {
    this.rules = rules;
    this.currentContext = null;
  }
  
  evaluateContext(applicationInfo, systemState) {
    const matchedRules = [];
    
    // Evaluate application-based rules
    for (const [appName, profile] of Object.entries(this.rules.applicationBased)) {
      if (this.matchesApplication(applicationInfo.processName, appName)) {
        matchedRules.push({ type: 'application', profile, priority: 100 });
      }
    }
    
    // Evaluate window title-based rules
    for (const rule of this.rules.windowTitleBased.patterns) {
      if (rule.regex.test(applicationInfo.windowTitle)) {
        matchedRules.push({ type: 'windowTitle', profile: rule.profile, priority: 90 });
      }
    }
    
    // Evaluate time-based rules
    const timeRule = this.evaluateTimeBasedRules(systemState.currentTime);
    if (timeRule) {
      matchedRules.push({ type: 'time', profile: timeRule.profile, priority: 50 });
    }
    
    // Return highest priority rule
    return matchedRules.length > 0 
      ? matchedRules.sort((a, b) => b.priority - a.priority)[0]
      : null;
  }
  
  matchesApplication(processName, ruleName) {
    // Fuzzy matching for application names
    const normalizedProcess = processName.toLowerCase().replace(/\s+/g, '');
    const normalizedRule = ruleName.toLowerCase().replace(/\s+/g, '');
    
    return normalizedProcess.includes(normalizedRule) || 
           normalizedRule.includes(normalizedProcess);
  }
}
```

  - Mitigation: Local-only processing and clear privacy documentation
- **Compatibility Risk**: API changes in OS updates may break detection
  - Mitigation: Version detection and graceful degradation strategies
